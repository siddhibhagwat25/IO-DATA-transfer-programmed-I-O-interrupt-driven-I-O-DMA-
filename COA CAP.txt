<script>
const method = document.getElementById("method");
const total = document.getElementById("total");
const startBtn = document.getElementById("start");
const pauseBtn = document.getElementById("pause");
const resetBtn = document.getElementById("reset");

const description = document.getElementById("description");
const cpuStatus = document.getElementById("cpuStatus");
const middleTitle = document.getElementById("middleTitle");
const middleStatus = document.getElementById("middleStatus");
const memoryStatus = document.getElementById("memoryStatus");
const flow = document.getElementById("flow");
const progress = document.getElementById("progress");
const progressText = document.getElementById("progressText");
const transferred = document.getElementById("transferred");
const busy = document.getElementById("busy");
const free = document.getElementById("free");
const blocks = document.getElementById("blocks");
const log = document.getElementById("log");

let count = 10;
let current = 0;
let busySteps = 0;
let freeSteps = 0;
let timer = null;
let running = false;
let paused = false;
let phase = 0;
let dmaRemaining = 0;

const descriptions = {
  pio: "Programmed I/O: The CPU repeatedly polls the device and copies data into memory.",
  interrupt: "Interrupt-driven I/O: The CPU does other work while waiting for a device interrupt, then copies the data.",
  dma: "DMA: The CPU initializes the DMA controller. DMA transfers blocks to memory while the CPU does other work."
};

function addLog(message) {
  const item = document.createElement("li");
  item.textContent = message;
  log.prepend(item);
}

function drawBlocks() {
  blocks.replaceChildren();

  for (let i = 0; i < count; i++) {
    const b = document.createElement("div");
    b.className = "block";
    if (i < current) b.classList.add("done");
    else if (i === current && running) b.classList.add("active");

    b.textContent = "Block " + (i + 1);
    blocks.appendChild(b);
  }
}

function render() {
  const pct = Math.round((current / count) * 100);

  progress.style.width = pct + "%";
  progressText.textContent = "Transfer Progress: " + pct + "%";
  transferred.textContent = current + " / " + count;
  busy.textContent = busySteps;
  free.textContent = freeSteps;
  memoryStatus.textContent = current + " block(s) received";

  drawBlocks();
}

function configure() {
  const m = method.value;

  description.textContent = descriptions[m];

  middleTitle.textContent =
    m === "dma" ? "DMA Controller" : "I/O Controller";

  flow.textContent =
    m === "dma"
      ? "I/O Device → DMA Controller → Main Memory"
      : "I/O Device → CPU → Main Memory";

  cpuStatus.textContent = "Idle";
  middleStatus.textContent = "Waiting";
}

function finish() {
  clearInterval(timer);
  timer = null;
  running = false;
  paused = false;

  startBtn.disabled = false;
  pauseBtn.disabled = true;

  cpuStatus.textContent = "Idle";
  middleStatus.textContent = "Transfer complete";

  addLog("Transfer completed: " + count + " data blocks.");
  render();
}

function step() {
  if (!running) return;

  const m = method.value;

  if (current >= count) {
    finish();
    return;
  }

  if (m === "pio") {
    // Every polling cycle occupies the CPU.
    busySteps++;
    cpuStatus.textContent = "Polling device";

    if (phase === 0) {
      middleStatus.textContent = "Device ready check";
      addLog("CPU polls the I/O device.");
      phase = 1;
    } else {
      middleStatus.textContent = "Sending block " + (current + 1);
      current++;
      phase = 0;
      addLog("CPU copies a data block to main memory.");
    }

  } else if (m === "interrupt") {
    if (phase === 0) {
      // CPU does other work while the device prepares data.
      freeSteps++;
      cpuStatus.textContent = "Executing other work";
      middleStatus.textContent = "Device preparing data";

      addLog("CPU continues other work while the device operates.");
      phase = 1;

    } else if (phase === 1) {
      busySteps++;
      cpuStatus.textContent = "Handling interrupt";
      middleStatus.textContent = "Interrupt requested";

      addLog("Device raises an interrupt; CPU starts its handler.");
      phase = 2;

    } else {
      busySteps++;
      cpuStatus.textContent = "Copying data";
      current++;

      middleStatus.textContent = "Next block";
      addLog("Interrupt handler copies a block to memory.");
      phase = 0;
    }

  } else {
    // DMA setup is a CPU operation performed once.
    if (phase === 0) {
      busySteps++;
      cpuStatus.textContent = "Configuring DMA";
      middleStatus.textContent = "DMA initialized";

      dmaRemaining = count;
      addLog("CPU configures DMA address, size and transfer direction.");
      phase = 1;

    } else if (dmaRemaining > 0) {
      // DMA moves data independently of the CPU.
      freeSteps++;
      cpuStatus.textContent = "Executing other work";
      middleStatus.textContent = "DMA transferring";

      current++;
      dmaRemaining--;

      addLog("DMA transfers block " + current + " directly to memory.");

      if (dmaRemaining === 0) {
        addLog("DMA signals transfer completion to the CPU.");
        cpuStatus.textContent = "Handling completion";
        middleStatus.textContent = "Transfer complete";
      }
    }
  }

  render();

  if (current >= count) {
    finish();
  }
}

function reset() {
  clearInterval(timer);
  timer = null;

  count = Number(total.value);
  current = 0;
  busySteps = 0;
  freeSteps = 0;
  phase = 0;
  dmaRemaining = 0;
  running = false;
  paused = false;

  startBtn.disabled = false;
  pauseBtn.disabled = true;

  log.replaceChildren();
  configure();
  render();

  addLog("Simulation initialized for " + count + " blocks.");
}

startBtn.addEventListener("click", () => {
  if (running) return;

  if (current >= count) reset();

  running = true;
  paused = false;

  startBtn.disabled = true;
  pauseBtn.disabled = false;

  addLog("Started " + method.options[method.selectedIndex].text + ".");

  timer = setInterval(step, 650);
  step();
});

pauseBtn.addEventListener("click", () => {
  if (!running) return;

  clearInterval(timer);
  timer = null;
  running = false;
  paused = true;

  startBtn.disabled = false;
  pauseBtn.disabled = true;
  cpuStatus.textContent = "Paused";

  addLog("Simulation paused.");
  render();
});

resetBtn.addEventListener("click", reset);

method.addEventListener("change", reset);
total.addEventListener("change", reset);

reset();
</script>
